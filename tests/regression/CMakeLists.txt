# Scientific case settings stay in YAML; CMake owns executable selection.
# The pinned NumPy version requires Python 3.11 or newer.
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import numpy, netCDF4, yaml"
  RESULT_VARIABLE _imports ERROR_VARIABLE _error)
if(NOT _imports EQUAL 0)
  message(FATAL_ERROR "Regression Python dependencies are missing: ${_error}")
endif()

set(CFBM_REFERENCE "" CACHE PATH "Explicit immutable regression reference")
set(CFBM_RUN_ROOT "${PROJECT_BINARY_DIR}/runs" CACHE PATH "Generated case directories")
set(CFBM_ESMX_EXECUTABLE "${CMAKE_INSTALL_PREFIX}/bin/esmx_fire" CACHE FILEPATH "ESMX application")
set(_runner "${CMAKE_CURRENT_SOURCE_DIR}/regression.py")
set(_drivers "")
option(CFBM_FULL_TESTS "Register the large scheduled HPC experiments" OFF)
if(CFBM_FULL_TESTS)
  list(APPEND _drivers --include-full)
endif()
if(NUOPC)
  list(APPEND _drivers --driver nuopc)
endif()
if(ESMX)
  list(APPEND _drivers --driver esmx)
endif()
if(DM_PARALLEL AND OPENMP)
  set(_variant hybrid)
elseif(DM_PARALLEL)
  set(_variant mpi)
elseif(OPENMP)
  set(_variant omp)
else()
  set(_variant serial)
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${_runner}" registrations
  --variant ${_variant} ${_drivers}
  OUTPUT_VARIABLE _tests RESULT_VARIABLE _status ERROR_VARIABLE _error)
if(NOT _status EQUAL 0)
  message(FATAL_ERROR "Cannot register regression tests: ${_error}")
endif()

function(add_cfbm_test record)
  foreach(_field name case scale configuration execution driver processors)
    string(JSON _${_field} GET "${record}" ${_field})
  endforeach()
  if(_driver STREQUAL standalone)
    set(_exe "$<TARGET_FILE:fire_behavior.exe>")
  elseif(_driver STREQUAL nuopc)
    set(_exe "$<TARGET_FILE:fire_behavior_esmf>")
  else()
    set(_exe "${CFBM_ESMX_EXECUTABLE}")
  endif()
  set(_arguments "")
  if(DM_PARALLEL)
    list(APPEND _arguments --launcher "${MPIEXEC_EXECUTABLE}"
      "--process-flag=${MPIEXEC_NUMPROC_FLAG}")
    foreach(_flag IN LISTS MPIEXEC_PREFLAGS)
      list(APPEND _arguments "--launcher-arg=${_flag}")
    endforeach()
    foreach(_flag IN LISTS MPIEXEC_POSTFLAGS)
      list(APPEND _arguments "--launcher-post-arg=${_flag}")
    endforeach()
  endif()
  if(CFBM_REFERENCE)
    list(APPEND _arguments --reference "${CFBM_REFERENCE}")
  endif()
  add_test(NAME ${_name} COMMAND "${Python3_EXECUTABLE}" -B "${_runner}" case
    --case ${_case} --scale ${_scale} --configuration ${_configuration}
    --driver ${_driver} --execution ${_execution} --executable "${_exe}"
    --run-root "${CFBM_RUN_ROOT}" ${_arguments})
  set(_labels "regression;case:${_case};driver:${_driver};execution:${_execution};configuration:${_configuration};scale:${_scale}")
  string(JSON _count LENGTH "${record}" labels)
  math(EXPR _last "${_count} - 1")
  foreach(_index RANGE ${_last})
    string(JSON _label GET "${record}" labels ${_index})
    list(APPEND _labels "${_label}")
  endforeach()
  set_tests_properties(${_name} PROPERTIES LABELS "${_labels}"
    PROCESSORS ${_processors} TIMEOUT 2400)
endfunction()

string(JSON _count LENGTH "${_tests}")
math(EXPR _last "${_count} - 1")
if(_count GREATER 0)
  foreach(_index RANGE ${_last})
    string(JSON _record GET "${_tests}" ${_index})
    add_cfbm_test("${_record}")
  endforeach()
endif()

add_test(NAME regression_python COMMAND "${Python3_EXECUTABLE}" -B -m unittest
  discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p "*_test.py" -v)
set_tests_properties(regression_python PROPERTIES
  LABELS "unit;quick;pr;full"
  ENVIRONMENT "CFBM_TEST_TMP=${PROJECT_BINARY_DIR}/python-test-artifacts")
