cmake_minimum_required(VERSION 3.20)
enable_testing()
add_subdirectory(unit)
set(CFBM_TEST_SYSTEM generated CACHE STRING "Test system: generated or legacy")
set_property(CACHE CFBM_TEST_SYSTEM PROPERTY STRINGS generated legacy)
if(CFBM_TEST_SYSTEM STREQUAL generated)
  add_subdirectory(regression)
elseif(CFBM_TEST_SYSTEM STREQUAL legacy)
  set(_legacy test7 test8)
  if(NUOPC)
    list(APPEND _legacy test7esmf test8esmf)
  endif()
  if(ESMX)
    list(APPEND _legacy test7esmx test8esmx testx)
  endif()
  foreach(_test IN LISTS _legacy)
    add_test(NAME ${_test} COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/run_legacy.sh"
      ${_test} "${CMAKE_INSTALL_PREFIX}" "${PROJECT_BINARY_DIR}/legacy-runs")
    set_tests_properties(${_test} PROPERTIES LABELS legacy TIMEOUT 1200)
  endforeach()
else()
  message(FATAL_ERROR "CFBM_TEST_SYSTEM must be generated or legacy")
endif()
